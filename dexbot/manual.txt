QUICK START
================================================================

  go run ./apps/start_all

  # Web dashboard (after starting daemons)
  http://localhost:8080

================================================================
GOVERNANCE DAEMON COMMANDS
================================================================

  # Show all daemon status (JSON)
  go run ./apps/governance -action=status

  # Show wallet balance (real BSC on-chain via MetaMask private key)
  go run ./apps/governance -action=balance

  # Reload logger configuration
  go run ./apps/governance -action=reload-log

  # Reload configuration from config.env (propagated to all daemons)
  go run ./apps/governance -action=reload-config

  # Restart a specific daemon
  go run ./apps/governance -action=restart -daemon=school

  # Stop a specific daemon
  go run ./apps/governance -action=stop -daemon=trading

  # Start a specific daemon
  go run ./apps/governance -action=start -daemon=school

  # Full system shutdown
  go run ./apps/governance -action=shutdown

  # Help — list all available commands
  go run ./apps/governance -action=help

  # Help — show configuration details
  go run ./apps/governance -action=help-configuration

  # Help — show verbose configuration
  go run ./apps/governance -action=help-configuration-vvv

  # Add a new token to the balance tracker
  go run ./apps/governance -action=addToken -name=MYTOKEN -address=0x...

  # Add a new blockchain network
  go run ./apps/governance -action=addChain -name=Ethereum -id=1 -url=https://mainnet.infura.io

================================================================
SCHOOL DAEMON COMMANDS
================================================================

  # Fetch last 5 real market data records
  go run ./apps/school -action=fetchMarket
